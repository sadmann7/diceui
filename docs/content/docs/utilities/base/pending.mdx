## Usage

### With Button

```tsx
import { Pending } from "@/components/pending"

export function SubmitButton() {
  const [isPending, startTransition] = useTransition()
  
  return (
    <Pending isPending={isPending} render={<button />}>
      Submit
    </Pending>
  )
}
```

### Using the Hook

```tsx
import { usePending } from "@/components/pending"

export function CustomButton() {
  const [isPending] = useTransition()
  const { pendingProps } = usePending({ isPending })
  
  return <button {...pendingProps}>Submit</button>
}
```

## Accessibility

When pending:
- Sets `aria-busy="true"` to indicate loading state
- Sets `aria-disabled="true"` to prevent interaction
- Prevents all pointer and keyboard events
- Maintains keyboard focus for better UX
